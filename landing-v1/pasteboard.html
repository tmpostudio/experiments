CUSTOM · ON-CHAINkamino_lending::cpi
// Compose Kamino Lend into your own program
use kamino_lending::cpi::{
    accounts::DepositReserveLiquidity, deposit,
};
 
pub fn sweep_to_kamino(
    ctx: Context<Sweep>, amount: u64,
) -> Result<()> {
  let accounts = DepositReserveLiquidity {
    owner: ctx.accounts.vault.to_account_info(),
    reserve: ctx.accounts.reserve.to_account_info(),
    lending_market: ctx.accounts.market(),
    ..ctx.accounts.kamino()
  };
  let program = ctx.accounts.klend();
  deposit(CpiContext::new(program, accounts), amount)
}
